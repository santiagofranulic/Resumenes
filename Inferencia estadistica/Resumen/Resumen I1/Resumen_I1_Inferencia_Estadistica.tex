\section{Estudios Estadísticos}

\begin{definicion}{Estudio Estadístico}{}
    Proceso de recolección, análisis e interpretación de datos.
\end{definicion}

\begin{definicion}{Tipos de Estudios Estadísticos}{}
    \begin{itemize}
        \item Segun el objetivo:\\
    \begin{tabular}{|l|l|}
        \hline
        \textbf{Analitico} & Busca explicar una relación causa-efecto entre variables.\\
        \hline
        \textbf{Descriptivo} & Describe una población o fenómeno.\\
        \hline
    \end{tabular}
    \vspace{0.1cm}
    \item Segun el periodo de tiempo:\\
    \begin{tabular}{|l|l|}
        \hline
        \textbf{Transversal} & Analiza en un unico momento en el tiempo.\\
        \hline
        \textbf{Longitudinal} & Analiza en distintos momentos en el tiempo.\\
        \hline
    \end{tabular}
    \vspace{0.1cm}
    \item Segun el control de variables:\\
    \begin{tabular}{|l|l|}
        \hline
        \textbf{Observacional} & No interviene el investigador.\\
        \hline
        \textbf{Experimental} & Interviene el investigador.\\
        \hline
    \end{tabular}
    \vspace{0.1cm}
    \item Segun la dirección temporal:\\
    \begin{tabular}{|l|l|}
        \hline
        \textbf{Prospectivo} & Analiza el fenómeno en el futuro.\\
        \hline
        \textbf{Retrospectivo} & Analiza el fenómeno en el pasado.\\
        \hline
    \end{tabular}
    \end{itemize}
\end{definicion}

\begin{definicion}{Recopilación de datos}{}
    \begin{itemize}
        \item Muestreo $\rightarrow$ Busca describir una poblacion.
        \begin{itemize}
            \item Probabilistico $\rightarrow$ Todos tienen una probabilidad de ser elegidos.
            \begin{itemize}
                \item Estratificado: Se divide la poblacion en estratos y se muestrea aleatoriamente de cada estrato.
                \item Conglomerados
                \item Multietápico
            \end{itemize}
            \item No probabilistico $\rightarrow$ No todos tienen una probabilidad de ser elegidos.
        \end{itemize}
        \item Reclutamiento $\rightarrow$ Busca explicar una relación causa-efecto entre variables.
    \end{itemize}
\end{definicion}

\section{Sesgos de Selección}

\begin{definicion}{Sesgos de Selección}{}
    Es el error de seleccionar una muestra no representativa de la población.
    \begin{nota}[Principio Base]
        Ningún experimento es perfecto, siempre habrá sesgos segun diseño, medición, análisis y muestra.
    \end{nota}
    \begin{itemize}
        \item Sesgos transversales: 
        \begin{itemize}
            \item Cobertura incompleta
            \item Participación desigual - No respuesta
            \item Selección no representativa
            \item Exclusion sistematica
        \end{itemize}
        \item Sesgos longitudinales:
        \begin{itemize}
            \item Perdida de seguimiento (sobretodo la perdida no aleatoria $\rightarrow$ sesgo del sobreviviente)
            \item Disponibilidad desigual de datos $\rightarrow$ no respuesta parcial o total
        \end{itemize}
    \end{itemize}
\end{definicion}

\section{Modelos Paramétricos}

\begin{definicion}{Modelos Paramétricos}{}
    Es una familia de distribuciones de probabilidad para una variable aleatoria $X$:
    \[ \mathcal{F} = \{ f(x|\theta) : \theta \in \Theta \} \quad \text{donde } \theta \in \Theta \text{ es un vector de parámetros.} \]
\end{definicion}

\begin{definicion}{Familia Exponencial}{}
    Es un modelo paramétrico que se puede escribir como:
    \[ f(x|\theta) = h(x) \exp{\left( \sum_{i=1}^s \eta_i(\theta) T_i(x) - B(\theta) \right)}\]
    O de forma equivalente:
    \[ f(x|\theta) = h(x) c(\theta) \exp{\langle \eta(\theta), T(x) \rangle}\]
    Esta familia comparte propiedades muy útiles para la inferencia estadística.
\end{definicion}

\begin{definicion}{Condiciones de Regularidad}{}
    \begin{itemize}
        \item El espacio de parámetros $\Theta$ es un conjunto abierto.
        \item El soporte $\mathcal{X}$ de la variable aleatoria no depende de $\theta$.
        \item Existen derivadas hasta segundo orden de $f(x|\theta)$ con respecto a $\theta$.
        \item Se puede intercambiar el orden de integración y derivación.
        \item Para todo $\theta \in \Theta$, $\int_{\mathcal{X}} f(x|\theta) dx = 1$.
    \end{itemize}
    \begin{nota}[Recordatorio]
        El soporte de una distribución de probabilidad es el conjunto de valores que la variable aleatoria puede tomar.
    \end{nota}
\end{definicion}

\section{Estimación de Parámetros}

\begin{definicion}{Estimadores Puntuales}{}
    Son funciones $\hat{\theta} = g(X_1, ..., X_n)$ de la muestra $X_1, ..., X_n$ que se utilizan para estimar el parámetro desconocido $\theta$.
\end{definicion}

\begin{definicion}{Función de Verosimilitud}{}
    Es una función que asigna un valor de probabilidad a cada posible valor del parámetro $\theta$, dado que se ha observado una muestra $X_1, ..., X_n$.
    \begin{equation}
        L(\theta | x_1, ..., x_n) = f(x_1, ..., x_n | \theta) = \prod_{i=1}^n f(x_i | \theta)
    \end{equation}
\end{definicion}

\begin{definicion}{Estimador de Máxima Verosimilitud}{EMV}
    Es el valor de $\theta$ que maximiza la función de verosimilitud.
    \begin{equation}
        \hat{\theta}_{EMV} = \arg\max_{\theta \in \Theta} L(\theta | x_1, ..., x_n)
    \end{equation}
    Metodo:
    \begin{enumerate}
        \item Calcular la función de verosimilitud $L(\theta | x_1, ..., x_n)$.
        \item Calcular la log-verosimilitud $l(\theta | x_1, ..., x_n) = \log L(\theta | x_1, ..., x_n)$. (Es mas facil derivar el logaritmo que la función original).
        \item Derivar la log-verosimilitud con respecto a $\theta$ y igualarla a 0.
        \item Resolver la ecuación para $\theta$.
    \end{enumerate}
    \begin{nota}[]
        La resolución siempre es el maximo global al operar con funciones concavas (log-verosimilitud).
    \end{nota}
\end{definicion}
